# Load an image from the path and call it 'ex'
load images/duck.jpg ex

# Compress the image using the Haar Wavelet Transform with a 30% compression level, name it 'ex-compressed', and save it
compress 30 ex ex-compressed30
save result/ex-compressed30.jpg ex-compressed30

# Compress the image using the Haar Wavelet Transform with a 30% compression level, name it 'ex-compressed', and save it
compress 90 ex ex-compressed90
save result/ex-compressed90.jpg ex-compressed90

# Histogram of the image and save it
histogram ex ex-histogram
save result/ex-histogram.jpg ex-histogram

# Flip the image vertically and call it 'ex-vertical' and save it
vertical-flip ex ex-vertical
save result/ex-vertical.jpg ex-vertical

# Flip the image horizontally and call it 'ex-horizontal' and save it
horizontal-flip ex ex-horizontal
save result/ex-horizontal.jpg ex-horizontal

# Flip the vertically flipped image horizontally and call it 'ex-vertical-horizontal' and save it
horizontal-flip ex-vertical ex-vertical-horizontal
save result/ex-vertical-horizontal.jpg ex-vertical-horizontal

# Create a red component image and call it 'ex-red' and save it
red-component ex ex-red
save result/ex-red.jpg ex-red

# Create a green component image and call it 'ex-green' and save it
green-component ex ex-green
save result/ex-green.jpg ex-green

# Create a blue component image and call it 'ex-blue' and save it
blue-component ex ex-blue
save result/ex-blue.jpg ex-blue

# Split the RGB components into three images: 'ex-red', 'ex-green', and 'ex-blue'
rgb-split ex ex-red ex-green ex-blue

# Combine the RGB images back into one and call it 'ex-combined'
rgb-combine ex-combined ex-red ex-green ex-blue
save result/ex-combined.jpg ex-combined

# Brighten the image by 20 units and call it 'ex-bright' and save it
brighten 20 ex ex-bright
save result/ex-bright.jpg ex-bright

# Darken the image by 20 units and call it 'ex-darkened' and save it
brighten -20 ex ex-darkened
save result/ex-darkened.jpg ex-darkened
